// =============================================================================

// AW MODEL - LIQUIDITY REVERSAL

// Implements the six-step reversal sequence:

//   1. Draw    -> nearest untaken swing liquidity (opposite side, used as target)

//   2. Neckline-> last opposing swing before the sweep

//   3. Sweep   -> wick through a swing high/low that closes back inside

//   4. Shift   -> displacement candle closing back through the neckline

//   5. Entry   -> retrace into the Fair Value Gap left by the displacement

//   6. Target  -> nearest untaken liquidity on the opposite side (or fixed R)

// Tracks both bullish and bearish reversals independently. Swing points confirm

// with a lag of `Swing Length` bars (normal for structure-based tools) - this

// is not a repainting signal generator, just a delayed structure confirmation.

// Also tracks each triggered trade to its resolution (stop or target hit) and

// keeps a running daily / cumulative P&L table.

//

// ---- CROSSTRADE SETUP (for the webhook alerts below) ----------------------

// CrossTrade is the bridge that turns a TradingView alert into a real order in

// NinjaTrader 8. Sign up here: https://crosstrade.io/pricing?ref=JNT

//   1. Create a CrossTrade account at the link above and pick a plan.

//   2. Inside CrossTrade, connect it to your NinjaTrader 8 install (it walks

//      you through installing its NT8 add-on).

//   3. In the CrossTrade dashboard, open your webhook/bridge settings and

//      copy your SECRET KEY - this is the value that goes in WEBHOOK_KEY

//      below, NOT your CrossTrade login password.

//   4. Paste that key between the quotes on the WEBHOOK_KEY line further

//      down in this script (search for "PASTE_YOUR_KEY_HERE"), directly in

//      Pine Editor's code view, then save.

//   5. In this indicator's Settings -> CrossTrade Webhook, fill in your NT8

//      Account name exactly as it appears in the CrossTrade/NT8 add-on, and

//      turn on "Send CrossTrade Webhook Alerts" when you're ready to go live.

//   6. On the chart, create ONE TradingView alert on this indicator with

//      condition "Any alert() function call", and paste the webhook URL

//      CrossTrade gives you into the alert's Notifications -> Webhook URL

//      field. That alert is what actually delivers every signal to CrossTrade.

// =============================================================================

// ---------------- Helpers ----------------

f_pad2(n) =>

    n < 10 ? "0" + str.tostring(n) : str.tostring(n)

f_dayKey() =>

    str.tostring(year(time)) + "-" + f_pad2(month(time)) + "-" + f_pad2(dayofmonth(time))

// Converts the chart's own contract symbol (e.g. MNQU2026) into NT8's expected

// short form (e.g. MNQU6 - root + month code + last digit of year). Falls back

// to the manual input if the ticker doesn't parse as expected (e.g. you're on

// a continuous MNQ1! chart, where this auto-detection cannot help).

f_autoInstrument() =>

    string result = manualInstrumentInput

    if str.length(syminfo.ticker) > str.length(syminfo.root)

        suffix = str.substring(syminfo.ticker, str.length(syminfo.root))

        if str.length(suffix) >= 2

            monthCode = str.substring(suffix, 0, 1)

            yearDigit = str.substring(suffix, str.length(suffix) - 1, str.length(suffix))

            result := syminfo.root + monthCode + yearDigit

    result

instrumentInput = autoInstrument ? f_autoInstrument() : manualInstrumentInput

// ---------------- Additional liquidity sources (HTF swings + prev-day H/L) ----------------

// Folded into the SAME liqHigh/liqLow arrays used above, so the existing

// "mark taken" loops and nearest-untaken target selection automatically cover

// these too - no separate tracking system needed.

f_pushLiquidity(array<float> arrPrice, array<int> arrBar, array<bool> arrTaken, float px) =>

    array.push(arrPrice, px)

    array.push(arrBar, bar_index)

    array.push(arrTaken, false)

    if array.size(arrPrice) > maxLiquidity

        array.shift(arrPrice)

        array.shift(arrBar)

        array.shift(arrTaken)

    true

htfAuto = _tfs <= 60 ? "15" : _tfs <= 180 ? "30" : _tfs <= 300 ? "60" : _tfs <= 900 ? "240" : _tfs <= 1800 ? "360" : _tfs <= 10800 ? "D" : _tfs <= 86400 ? "W" : "3M"

      ta.valuewhen(not na(ta.pivothigh(htfPivotStrength, htfPivotStrength)), time[htfPivotStrength], 0),

      ta.valuewhen(not na(ta.pivotlow(htfPivotStrength, htfPivotStrength)), ta.pivotlow(htfPivotStrength, htfPivotStrength), 0),

// confirmed data via lookahead_on since it only ever references a fully closed prior day.

[pdh, pdl, pdt] = request.security(syminfo.tickerid, "D", [high[1], low[1], time[1]], lookahead = barmerge.lookahead_on)

if barstate.isconfirmed and pdOk and not na(pdt)

    if na(pdt[1]) or pdt != pdt[1]

        f_pushLiquidity(liqHighPrice, liqHighBar, liqHighTaken, pdh)

        f_pushLiquidity(liqLowPrice, liqLowBar, liqLowTaken, pdl)

f_nearestUntakenAbove(price) =>

    float result = na

    if array.size(liqHighPrice) > 0

        for i = 0 to array.size(liqHighPrice) - 1

            p = array.get(liqHighPrice, i)

            taken = array.get(liqHighTaken, i)

            if not taken and p > price and (na(result) or p < result)

                result := p

    result

f_nearestUntakenBelow(price) =>

    float result = na

    if array.size(liqLowPrice) > 0

        for i = 0 to array.size(liqLowPrice) - 1

            p = array.get(liqLowPrice, i)

            taken = array.get(liqLowTaken, i)

            if not taken and p < price and (na(result) or p > result)

                result := p

    result

// ---------------- P&L tracking ----------------

var map<string, float> dailyPnl = map.new<string, float>()

var map<string, int> dailyWins = map.new<string, int>()

var map<string, int> dailyLosses = map.new<string, int>()

var array<string> dayOrder = array.new<string>(0)

// cumTotal and win/loss counters are wrapped in arrays because a function

// cannot reassign a plain global `var` primitive in Pine (CE10088) - mutating

// array CONTENTS from inside a function is legal, so this is the standard

// workaround. winLossArr: [0] = total wins, [1] = total losses.

var array<float> cumTotalArr = array.new<float>(1, 0.0)

var array<int> winLossArr = array.new<int>(2, 0)

f_recordPnl(points) =>

    dollars = points * pointValue

    array.set(cumTotalArr, 0, array.get(cumTotalArr, 0) + dollars)

    isWin = points > 0

    wlIdx = isWin ? 0 : 1

    array.set(winLossArr, wlIdx, array.get(winLossArr, wlIdx) + 1)

    key = f_dayKey()

    if map.contains(dailyPnl, key)

        map.put(dailyPnl, key, map.get(dailyPnl, key) + dollars)

        map.put(dailyWins, key, map.get(dailyWins, key) + (isWin ? 1 : 0))

        map.put(dailyLosses, key, map.get(dailyLosses, key) + (isWin ? 0 : 1))

    else

        map.put(dailyPnl, key, dollars)

        map.put(dailyWins, key, isWin ? 1 : 0)

        map.put(dailyLosses, key, isWin ? 0 : 1)

        array.push(dayOrder, key)

    dollars

// ---------------- Strategy Tester bridge ----------------
// Signal logic, invalidations, stop and target calculations below are unchanged.
// The original indicator records a theoretical fill at the already-known low/high

var bool bullActive = false

sweepLowCond = not bullActive and not na(lastPivotLow) and low < lastPivotLow and close > lastPivotLow and bar_index > lastPivotLowBar

if sweepLowCond

    bullActive := true

    bullMSS := false

    bullFVGSet := false

    bullTriggered := false

    bullSweptLow := low

    bullSweepBar := bar_index

    bullNeckline := lastPivotHigh

    bullNecklineBar := lastPivotHighBar

    if showSweeps

        label.new(bar_index, low, "SWEEP", style=label.style_label_up, color=bullColor, textcolor=color.white, size=size.tiny)

    if showNeckline and not na(bullNeckline)

        bullNecklineLine := line.new(bullNecklineBar, bullNeckline, bar_index + maxBarsToMSS, bullNeckline, color=bullColor, style=line.style_dashed)

if bullActive and not bullMSS

    if bar_index - bullSweepBar > maxBarsToMSS

        bullActive := false

    else if not na(bullNeckline) and close > bullNeckline and (close - open) >= atrVal * dispMult and close > open

        bullMSS := true

        bullMSSBar := bar_index

        if showNeckline and not na(bullNecklineLine)

            line.set_x2(bullNecklineLine, bar_index)

        if showSweeps

            label.new(bar_index, high, "SHIFT", style=label.style_label_down, color=bullColor, textcolor=color.white, size=size.tiny)

if bullActive and bullMSS and not bullFVGSet

    if bar_index - bullMSSBar > fvgSearchWindow + 2

        bullActive := false

    else if low > high[2]

        bullFVGSet := true

        bullFVGTop := low

        bullFVGBot := high[2]

        bullFVGBar := bar_index

        if showFVG

            bullFVGBox := box.new(bar_index[2], bullFVGBot, bar_index + maxBarsToEntry, bullFVGTop, border_color=bullColor, bgcolor=color.new(bullColor, 85))

if bullActive and bullFVGSet and not bullTriggered

    if bar_index - bullFVGBar > maxBarsToEntry

        bullActive := false

    else if close < bullSweptLow

        bullActive := false

    else if low <= bullFVGTop and not liveLongOpen and sessionAllowed

        bullTriggered := true

        entryPriceBull = math.min(bullFVGTop, high)

        stopBull = bullSweptLow

        targetBull = float(na)

        if targetMode == "Opposite Liquidity"

            tAbove = f_nearestUntakenAbove(close)

            targetBull := na(tAbove) ? entryPriceBull + (entryPriceBull - stopBull) * targetR : tAbove

        else

            targetBull := entryPriceBull + (entryPriceBull - stopBull) * targetR

        liveLongOpen := true

        liveLongEntry := entryPriceBull

        liveLongStop := stopBull

        liveLongTarget := targetBull

        // Strategy Tester order: same entry SIGNAL bar, exact original SL/TP levels.
        if strategy.position_size == 0 and not reportOrderIssuedThisBar
            strategy.entry("AW Long", strategy.long)
            strategy.exit("AW Long Exit", from_entry="AW Long", stop=stopBull, limit=targetBull)
            reportOrderIssuedThisBar := true

        if showTrade

            label.new(bar_index, low, "BUY\n" + str.tostring(math.round_to_mintick(entryPriceBull)), style=label.style_label_up, color=bullColor, textcolor=color.white, size=size.small)

            liveLongStopLine := line.new(bar_index, stopBull, bar_index + projectBars, stopBull, color=color.red, style=line.style_dotted)

            liveLongTargetLine := line.new(bar_index, targetBull, bar_index + projectBars, targetBull, color=color.green, style=line.style_dotted)

        isFirstBull = f_dayKey() != lastEntryDayKey

        f_sendEntryWebhook("BUY", stopBull, targetBull, isFirstBull)

        lastEntryDayKey := f_dayKey()

        bullEntrySignal := true

        bullActive := false

var bool bearActive = false

var bool bearMSS = false

var bool bearFVGSet = false

var bool bearTriggered = false

var float bearNeckline = na

var int bearNecklineBar = na

var float bearSweptHigh = na

var int bearSweepBar = na

var int bearMSSBar = na

var float bearFVGTop = na

var float bearFVGBot = na

var int bearFVGBar = na

var line bearNecklineLine = na

var box bearFVGBox = na

sweepHighCond = not bearActive and not na(lastPivotHigh) and high > lastPivotHigh and close < lastPivotHigh and bar_index > lastPivotHighBar

if sweepHighCond

    bearActive := true

    bearMSS := false

    bearFVGSet := false

    bearTriggered := false

    bearSweptHigh := high

    bearSweepBar := bar_index

    bearNeckline := lastPivotLow

    bearNecklineBar := lastPivotLowBar

    if showSweeps

        label.new(bar_index, high, "SWEEP", style=label.style_label_down, color=bearColor, textcolor=color.white, size=size.tiny)

    if showNeckline and not na(bearNeckline)

        bearNecklineLine := line.new(bearNecklineBar, bearNeckline, bar_index + maxBarsToMSS, bearNeckline, color=bearColor, style=line.style_dashed)

if bearActive and not bearMSS

    if bar_index - bearSweepBar > maxBarsToMSS

        bearActive := false

    else if not na(bearNeckline) and close < bearNeckline and (open - close) >= atrVal * dispMult and close < open

        bearMSS := true

        bearMSSBar := bar_index

        if showNeckline and not na(bearNecklineLine)

            line.set_x2(bearNecklineLine, bar_index)

        if showSweeps

            label.new(bar_index, low, "SHIFT", style=label.style_label_up, color=bearColor, textcolor=color.white, size=size.tiny)

if bearActive and bearMSS and not bearFVGSet

    if bar_index - bearMSSBar > fvgSearchWindow + 2

        bearActive := false

    else if high < low[2]

        bearFVGSet := true

        bearFVGTop := low[2]

        bearFVGBot := high

        bearFVGBar := bar_index

        if showFVG

            bearFVGBox := box.new(bar_index[2], bearFVGBot, bar_index + maxBarsToEntry, bearFVGTop, border_color=bearColor, bgcolor=color.new(bearColor, 85))

if bearActive and bearFVGSet and not bearTriggered

    if bar_index - bearFVGBar > maxBarsToEntry

        bearActive := false

    else if close > bearSweptHigh

        bearActive := false

    else if high >= bearFVGBot and not liveShortOpen and sessionAllowed

        bearTriggered := true

        entryPriceBear = math.max(bearFVGBot, low)

        stopBear = bearSweptHigh

        targetBear = float(na)

        if targetMode == "Opposite Liquidity"

            tBelow = f_nearestUntakenBelow(close)

            targetBear := na(tBelow) ? entryPriceBear - (stopBear - entryPriceBear) * targetR : tBelow

        else

            targetBear := entryPriceBear - (stopBear - entryPriceBear) * targetR

        liveShortOpen := true

        liveShortEntry := entryPriceBear

        liveShortStop := stopBear

        liveShortTarget := targetBear

        // Strategy Tester order: same entry SIGNAL bar, exact original SL/TP levels.
        if strategy.position_size == 0 and not reportOrderIssuedThisBar
            strategy.entry("AW Short", strategy.short)
            strategy.exit("AW Short Exit", from_entry="AW Short", stop=stopBear, limit=targetBear)
            reportOrderIssuedThisBar := true

        if showTrade

            label.new(bar_index, high, "SELL\n" + str.tostring(math.round_to_mintick(entryPriceBear)), style=label.style_label_down, color=bearColor, textcolor=color.white, size=size.small)

            liveShortStopLine := line.new(bar_index, stopBear, bar_index + projectBars, stopBear, color=color.red, style=line.style_dotted)

            liveShortTargetLine := line.new(bar_index, targetBear, bar_index + projectBars, targetBear, color=color.green, style=line.style_dotted)

        isFirstBear = f_dayKey() != lastEntryDayKey

        f_sendEntryWebhook("SELL", stopBear, targetBear, isFirstBear)

        lastEntryDayKey := f_dayKey()

        bearEntrySignal := true

        bearActive := false

// ---------------- Plots / Alerts ----------------

plotshape(bullEntrySignal, title="Bull Entry", location=location.belowbar, style=shape.triangleup, color=bullColor, size=size.small)

plotshape(bearEntrySignal, title="Bear Entry", location=location.abovebar, style=shape.triangledown, color=bearColor, size=size.small)

alertcondition(bullEntrySignal, title="AW Model Bullish Entry", message="AW Model: bullish sweep + shift + FVG entry triggered")

alertcondition(bearEntrySignal, title="AW Model Bearish Entry", message="AW Model: bearish sweep + shift + FVG entry triggered")

// ---------------- P&L Table ----------------

tablePosVal = tablePos == "top_right" ? position.top_right : tablePos == "top_left" ? position.top_left : tablePos == "bottom_right" ? position.bottom_right : position.bottom_left

var table pnlTable = table.new(tablePosVal, 4, maxTableRows + 2, border_width=1)

f_fmtDollars(v) =>

    (v >= 0 ? "$" : "-$") + str.tostring(math.round(math.abs(v) * 100) / 100)

// Solid (low-transparency) fills so the table reads clearly regardless of

// chart background - Dark theme for a black/dark chart, Light theme for a

// white/light chart. Text color flips with it so it never washes out.

isDarkTheme = tableTheme == "Dark"

headerBg = isDarkTheme ? color.new(#1b1b1b, 0) : color.new(#d6d6d6, 0)

headerText = isDarkTheme ? color.white : color.black

dateBg = isDarkTheme ? color.new(#2b2b2b, 0) : color.new(#f0f0f0, 0)

neutralText = isDarkTheme ? color.white : color.black

winBg = isDarkTheme ? color.new(#0f4d2e, 0) : color.new(#c8ecd9, 0)

lossBg = isDarkTheme ? color.new(#5c1414, 0) : color.new(#f6cccc, 0)

winText = isDarkTheme ? color.white : color.black

lossText = isDarkTheme ? color.white : color.black

totalBg = isDarkTheme ? color.new(#3a3a3a, 0) : color.new(#c2c2c2, 0)

if showTable and barstate.islast

    table.clear(pnlTable, 0, 0, 3, maxTableRows + 1)

    table.cell(pnlTable, 0, 0, "Date", bgcolor=headerBg, text_color=headerText)

    table.cell(pnlTable, 1, 0, "Daily $", bgcolor=headerBg, text_color=headerText)

    table.cell(pnlTable, 2, 0, "Win Rate", bgcolor=headerBg, text_color=headerText)

    table.cell(pnlTable, 3, 0, "Cum. $", bgcolor=headerBg, text_color=headerText)

    n = array.size(dayOrder)

    startIdx = math.max(0, n - maxTableRows)

    runningCum = 0.0

    // pre-sum any days before the displayed window so the cumulative column stays correct

    if startIdx > 0

        for i = 0 to startIdx - 1

            runningCum := runningCum + map.get(dailyPnl, array.get(dayOrder, i))

    row = 1

    if n > 0

        for i = startIdx to n - 1

            key = array.get(dayOrder, i)

            dayVal = map.get(dailyPnl, key)

            dWins = map.get(dailyWins, key)

            dLosses = map.get(dailyLosses, key)

            dTrades = dWins + dLosses

            dWinRate = dTrades > 0 ? math.round(dWins / dTrades * 100) : 0

            runningCum := runningCum + dayVal

            dayBg = dayVal >= 0 ? winBg : lossBg

            dayTxt = dayVal >= 0 ? winText : lossText

            cumBg = runningCum >= 0 ? winBg : lossBg

            cumTxt = runningCum >= 0 ? winText : lossText

            table.cell(pnlTable, 0, row, key, bgcolor=dateBg, text_color=neutralText)

            table.cell(pnlTable, 1, row, f_fmtDollars(dayVal), bgcolor=dayBg, text_color=dayTxt)

            table.cell(pnlTable, 2, row, str.tostring(dWins) + "-" + str.tostring(dLosses) + " (" + str.tostring(dWinRate) + "%)", bgcolor=dateBg, text_color=neutralText)

            table.cell(pnlTable, 3, row, f_fmtDollars(runningCum), bgcolor=cumBg, text_color=cumTxt)

            row += 1

    cumTotal = array.get(cumTotalArr, 0)

    totalWins = array.get(winLossArr, 0)

    totalLosses = array.get(winLossArr, 1)

    totalTrades = totalWins + totalLosses

    totalWinRate = totalTrades > 0 ? math.round(totalWins / totalTrades * 100) : 0

    totalValBg = cumTotal >= 0 ? winBg : lossBg

    totalValTxt = cumTotal >= 0 ? winText : lossText

    table.cell(pnlTable, 0, row, "TOTAL", bgcolor=totalBg, text_color=neutralText)

    table.cell(pnlTable, 1, row, "", bgcolor=totalBg)

    table.cell(pnlTable, 2, row, str.tostring(totalWins) + "-" + str.tostring(totalLosses) + " (" + str.tostring(totalWinRate) + "%)", bgcolor=totalBg, text_color=neutralText)

    table.cell(pnlTable, 3, row, f_fmtDollars(cumTotal), bgcolor=totalValBg, text_color=totalValTxt)
